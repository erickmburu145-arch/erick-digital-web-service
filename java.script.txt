// Welcome message in the browser console
console.log("Welcome to Erick Digital Services!");

// Display a message when a service is selected
const serviceCards = document.querySelectorAll(".card");

serviceCards.forEach(function(card) {
  card.addEventListener("click", function() {
    const serviceName = card.querySelector("h3").textContent;

    alert(
      "Thank you for your interest in " +
      serviceName +
      "! Contact Erick Digital Services for more information."
    );
  });
});

// Highlight the active navigation link
const navigationLinks = document.querySelectorAll("nav a");

navigationLinks.forEach(function(link) {
  link.addEventListener("click", function() {
    navigationLinks.forEach(function(item) {
      item.style.color = "white";
    });

    link.style.color = "#36a9ff";
  });
});
